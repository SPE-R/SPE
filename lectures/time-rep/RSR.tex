\begin{frame}{Cohort rates vs. population rates: RSR}
  \begin{itemize}[<+->]
  \item \textbf{Additive:} $\lambda(a) = \delta(a) +
    \lambda_\text{p}(a)$, \quad $\lambda_\text{p}$ assumed known
  \item Note that the survival (since $a=a_0$, say) is: \vspace*{-1ex}
\begin{align*}
     S(a) &= \exp\Bigl(-\!\!\int_{a_0}^a\!\! \delta(a) + \lambda_\text{p}(a) \dif a\Bigr)\\
          &= \exp\Bigl(-\!\!\int_{a_0}^a\!\! \delta(a)          \dif a\Bigr) \times
             \exp\Bigl(-\!\!\int_{a_0}^a\!\! \lambda_\text{p}(a) \dif a\Bigr)
          &= \exp\Bigl(-\!\!  \int_{a_0}^a\!\! \delta(a) \dif a\Bigr) \times S_\text{p}(a)\\
    \pause \Rightarrow \quad
    r(a) = S(a)/S_\text{p}(a) &= \exp\Bigl(-\!\!\int_{a_0}^a\!\! \delta(a) \dif a\Bigr)
  \end{align*}
    \pause
  \item \textbf{Additive} model for \textbf{rates} $\Leftrightarrow$ \textbf{Relative survival} model.
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{Simple additive model}
A model where the \textbf{difference} between the cohort and the
population rates are the parameters.

\pause
Poisson model with \textcolor{forestgreen}{identity link}
and the expected numbers as \textcolor{forestgreen}{offset}

\pause
\ldots not easy to fit:
\begin{Schunk}
\begin{Sinput}
> try(
+ glm(cbind(lex.Xst == 1, lex.dur) ~ factor(contrast) - 1,
+     offset = E,
+     family = poisreg(link = "identity"),
+     data = thadx) ) ; geterrmessage() ; warnings()
\end{Sinput}

\end{Soutput}

\end{frame}
